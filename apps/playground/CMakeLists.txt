cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(openfactory_playground)
target_sources(app PRIVATE src/main.c src/monitor.c src/commands.c)
foreach(name main.c prj.conf README.txt)
  if(name STREQUAL "main.c")
    set(source ${CMAKE_CURRENT_SOURCE_DIR}/src/${name})
  else()
    set(source ${CMAKE_CURRENT_SOURCE_DIR}/${name})
  endif()
  generate_inc_file_for_target(app ${source} ${ZEPHYR_BINARY_DIR}/include/generated/${name}.inc)
endforeach()
